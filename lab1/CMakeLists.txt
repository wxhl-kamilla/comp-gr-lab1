cmake_minimum_required(VERSION 3.22)
include(FetchContent)
project(vulkan-starter-app LANGUAGES C CXX)
add_executable(${PROJECT_NAME} source/main.cpp source/graphics_internal.cpp source/application.cpp)
set_target_properties(${PROJECT_NAME} PROPERTIES CXX_STANDARD_REQUIRED TRUE CXX_STANDARD 20)
if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /W4 /wd4201 /utf-8)
    target_link_options(${PROJECT_NAME} PRIVATE /subsystem:windows /entry:mainCRTStartup)
    target_compile_definitions(${PROJECT_NAME} PRIVATE _USE_MATH_DEFINES)
else()
    target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
endif()
# Библиотеки из рабочей версии ноутбука общие для всех трёх работ.
set(LAB_EXTERNAL "${CMAKE_CURRENT_SOURCE_DIR}/../external")
FetchContent_Declare(glfw SOURCE_DIR "${LAB_EXTERNAL}/glfw-master")
FetchContent_Declare(vk-bootstrap SOURCE_DIR "${LAB_EXTERNAL}/vk-bootstrap")
FetchContent_Declare(VulkanMemoryAllocator SOURCE_DIR "${LAB_EXTERNAL}/VulkanMemoryAllocator")
FetchContent_Declare(imgui SOURCE_DIR "${LAB_EXTERNAL}/imgui")
find_package(Vulkan REQUIRED)
set(GLFW_LIBRARY_TYPE STATIC)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
endif()
FetchContent_MakeAvailable(glfw vk-bootstrap VulkanMemoryAllocator imgui)
target_link_libraries(${PROJECT_NAME} PRIVATE glfw Vulkan::Vulkan vk-bootstrap::vk-bootstrap GPUOpen::VulkanMemoryAllocator)
target_include_directories(${PROJECT_NAME} PRIVATE "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends" "${LAB_EXTERNAL}/stb")
target_sources(${PROJECT_NAME} PRIVATE
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_vulkan.cpp
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/imgui_demo.cpp)
find_program(GLSLC glslc HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/bin" REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_compile_definitions(${PROJECT_NAME} PRIVATE GLFW_INCLUDE_NONE
    LAB_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders/"
    LAB_TEXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/textures/")
set(SHADER_BINARIES)
foreach(SHADER box.vert box.frag)
    set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}")
    set(BINARY "${CMAKE_CURRENT_BINARY_DIR}/shaders/${SHADER}.spv")
    add_custom_command(OUTPUT "${BINARY}" COMMAND "${GLSLC}" "${SOURCE}" -o "${BINARY}" DEPENDS "${SOURCE}" VERBATIM)
    list(APPEND SHADER_BINARIES "${BINARY}")
endforeach()
add_custom_target(shaders DEPENDS ${SHADER_BINARIES})
add_dependencies(${PROJECT_NAME} shaders)
set_property(TARGET ${PROJECT_NAME} PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
